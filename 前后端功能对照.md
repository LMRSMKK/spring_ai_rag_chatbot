#### （3）新建 Service 方法 `ChatService.java`（新建文件）

```java
package com.example.spring_ai_language.service;

import com.example.spring_ai_language.mapper.ChatMessageMapper;
import com.example.spring_ai_language.Uitl.SessionSummary;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Autowired;
import org.springframework.stereotype.Service;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.LocalTime;
import java.time.temporal.ChronoUnit;
import java.util.ArrayList;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.stream.Collectors;

@Service
@Slf4j
public class ChatService {

    @Autowired
    private ChatMessageMapper chatMessageMapper;

    /**
     * 返回带分组的会话列表，供前端侧边栏渲染
     */
    public List<Map<String, Object>> listSessions(int page, int size) {
        int offset = page * size;
        List<SessionSummary> summaries = buildSummaries(offset, size);

        // 按分组归类
        Map<String, List<SessionSummary>> grouped = new HashMap<>();
        LocalDate today = LocalDate.now();
        for (SessionSummary s : summaries) {
            String label;
            if (s.getLastMessageTime().toLocalDate().isEqual(today)) {
                label = "今天";
            } else if (s.getLastMessageTime().toLocalDate().isEqual(today.minusDays(1))) {
                label = "昨天";
            } else {
                label = "最近7天";
            }
            grouped.computeIfAbsent(label, k -> new ArrayList<>()).add(s);
        }

        // 保持固定顺序
        List<Map<String, Object>> result = new ArrayList<>();
        for (String key : List.of("今天", "昨天", "最近7天")) {
            List<SessionSummary> items = grouped.getOrDefault(key, List.of());
            if (!items.isEmpty()) {
                Map<String, Object> group = new HashMap<>();
                group.put("label", key);
                group.put("items", items);
                result.add(group);
            }
        }
        return result;
    }

    private List<SessionSummary> buildSummaries(int offset, int size) {
        // 查最近的 distinct session 及其最新一条消息
        List<ChatMessage> rows = chatMessageMapper.selectList(
                com.baomidou.mybatisplus.core.conditions.query.LambdaQueryWrapper.<ChatMessage>lambdaOrder()
                        .orderByDesc(ChatMessage::getCreateTime)
                        .last("LIMIT " + offset + "," + size));

        // 按 sessionId 去重，取第一条（即最新的）
        Map<String, ChatMessage> latestMap = rows.stream()
                .collect(Collectors.toMap(
                        ChatMessage::getSessionId,
                        m -> m,
                        (a, b) -> a.getCreateTime().isAfter(b.getCreateTime()) ? a : b));

        List<SessionSummary> result = new ArrayList<>();
        for (ChatMessage m : latestMap.values()) {
            SessionSummary s = new SessionSummary();
            s.setSessionId(m.getSessionId());
            s.setTitle(truncate(m.getContent(), 30));
            s.setLastMessageTime(m.getCreateTime());
            result.add(s);
        }
        return result;
    }

    private String truncate(String text, int maxLen) {
        if (text == null) return "";
        return text.length() <= maxLen ? text : text.substring(0, maxLen) + "...";
    }
}
```

> 注：上面用了 MyBatisPlus 的 `lambdaOrder()`，如需兼容旧版可改用 XML SQL。

```java
package com.example.spring_ai_language.mapper;

import com.baomidou.mybatisplus.core.mapper.BaseMapper;
import com.example.spring_ai_language.entity.KbDocument;
import org.apache.ibatis.annotations.Mapper;
import org.apache.ibatis.annotations.Select;

import java.util.List;
import java.util.Map;

@Mapper
public interface KbDocumentMapper extends BaseMapper<KbDocument> {

    /**
     * 统计每个知识库的文档数和切片数
     */
    @Select("SELECT d.knowledge_base_id AS kbId," +
            "       COUNT(DISTINCT d.id) AS docCount," +
            "       COUNT(c.id) AS chunkCount" +
            "  FROM kb_document d" +
            "  LEFT JOIN kb_chunk c ON c.doc_id = d.id" +
            " GROUP BY d.knowledge_base_id")
    List<Map<String, Object>> countByKnowledgeBase();

    /**
     * 获取某知识库下的文件名列表（用于右侧面板"已上传文档"）
     */
    @Select("SELECT file_name AS fileName, status, upload_time AS uploadTime" +
            "  FROM kb_document" +
            " WHERE knowledge_base_id = #{kbId}" +
            " ORDER BY upload_time DESC")
    List<Map<String, Object>> listFilesByKb(@Param("kbId") String kbId);
}
```

```java
@Autowired
private KbDocumentMapper kbDocumentMapper;

@GetMapping("/knowledge-bases")
public Result<?> listKnowledgeBases() {
    List<Map<String, Object>> stats = kbDocumentMapper.countByKnowledgeBase();
    List<Map<String, Object>> result = new ArrayList<>();
    for (Map<String, Object> row : stats) {
        String kbId = row.get("kbId").toString();
        List<Map<String, Object>> files = kbDocumentMapper.listFilesByKb(kbId);
        Map<String, Object> kb = new HashMap<>();
        kb.put("kbId", kbId);
        kb.put("docCount", row.get("docCount"));
        kb.put("chunkCount", row.get("chunkCount"));
        kb.put("files", files.stream()
                .map(f -> f.get("fileName").toString())
                .collect(Collectors.toList()));
        result.add(kb);
    }
    return Result.success(result);
}
```

```java
@Autowired
private RedisTemplate<String, String> redisTemplate;

private static final int SHARE_EXPIRE_DAYS = 7;

@GetMapping("/share/{sessionId}")
public Result<?> getShareSession(@PathVariable String sessionId) {
    String sharedKey = "share:" + sessionId;
    String storedSessionId = redisTemplate.opsForValue().get(sharedKey);
    if (storedSessionId == null) {
        return Result.error(404, "分享链接已过期或不存在");
    }
    // 只读返回该 session 的消息
    List<ChatMessage> messages = chatMessageMapper.selectAllBySession(storedSessionId, 100);
    List<Map<String, Object>> result = messages.stream().map(m -> {
        Map<String, Object> map = new HashMap<>();
        map.put("role", m.getRole());
        map.put("content", m.getContent());
        map.put("time", m.getCreateTime().toString());
        return map;
    }).collect(Collectors.toList());
    return Result.success(result);
}

@PostMapping("/share")
public Result<?> createShareLink(@RequestBody Map<String, String> body) {
    String sessionId = body.get("sessionId");
    if (sessionId == null) return Result.error(400, "sessionId 不能为空");
    String shareToken = UUID.randomUUID().toString().replace("-", "").substring(0, 12);
    String shareKey = "share:" + shareToken;
    redisTemplate.opsForValue().set(shareKey, sessionId, SHARE_EXPIRE_DAYS, TimeUnit.DAYS);
    String url = "https://" + getHost() + "/share?s=" + shareToken;
    return Result.success(Map.of("url", url));
}

private String getHost() {
    ServletRequestAttributes attrs = (ServletRequestAttributes) RequestContextHolder.getRequestAttributes();
    if (attrs != null) {
        return attrs.getRequest().getHeader("Host");
    }
    return "localhost:8080";
}
```

```javascript
  formData.append('knowledgeBaseId', kbId);
  formData.append('userId', userId);
```

```javascript
        const data = JSON.parse(xhr.responseText);
        onComplete(data);
      } else {
        onError(new Error(`HTTP ${xhr.status}`));
      }
    }
  };
  xhr.open('POST', '/api/document/upload');
  xhr.send(formData);
```

```java
private String extractUserId() {
    ServletRequestAttributes attrs =
            (ServletRequestAttributes) RequestContextHolder.getRequestAttributes();
    if (attrs != null) {
        // 优先读 Header
        String uid = attrs.getRequest().getHeader("X-User-Id");
        if (uid != null && !uid.isBlank()) return uid;
        // 其次读 Request Body 已放入 attribute 的值
        String bodyUid = (String) attrs.getRequest().getAttribute("userId");
        if (bodyUid != null && !bodyUid.isBlank()) return bodyUid;
    }
    return "anonymous";
}
```
